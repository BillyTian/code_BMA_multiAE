# JAGS model: multinomial AE counts with censoring-aware likelihood terms.
# The suffixes 11, 10, 01, and 00 indicate whether grade 1-5 and grade 3-5
# counts are observed (1) or censored (0), in that order.
model
{
  # Prior distributions
  tau_study1 ~ dt(0, 1, 1)T(0,)
  tau_study2 ~ dt(0, 1, 1)T(0,)
  tau_INT1 ~ dt(0, 1, 1)T(0,)
  tau_INT2 ~ dt(0, 1, 1)T(0,)

  mu_study1 ~ dnorm(0, 1/10^2)
  mu_study2 ~ dnorm(0, 1/10^2)

  # Horseshoe priors for AE-specific log odds ratios
  tau15 ~ dt(0, 1, 1)T(0,)
  tau35 ~ dt(0, 1, 1)T(0,)
  for (k in 1:n_AE){
    lambda15[k] ~ dt(0, 1, 1)T(0,)
    lambda35[k] ~ dt(0, 1, 1)T(0,)
    alpha15[k] ~ dnorm(0, pow(lambda15[k]*tau15, -2))T(-5,5)
    alpha35[k] ~ dnorm(0, pow(lambda35[k]*tau35, -2))T(-5,5)

    OR15[k] <- exp( alpha15[k] )
    OR35[k] <- exp( alpha35[k] )

    u_AE1[k] ~ dnorm(0, 1/10^2)
    u_AE2[k] ~ dnorm(0, 1/10^2)

    for (i in 1:n_study){
      u_INT1[i, k] ~ dnorm(u_AE1[k], pow(tau_INT1, -2))
      u_INT2[i, k] ~ dnorm(u_AE2[k], pow(tau_INT2, -2))
    }

    r1A_AE[k] <- exp( mu_study1 + u_AE1[k] )
    r2A_AE[k] <- exp( mu_study2 + u_AE2[k] )
    p35A_AE[k] <- r2A_AE[k]/(1+r1A_AE[k]+r2A_AE[k])
    p15A_AE[k] <- (r1A_AE[k]+r2A_AE[k])/(1+r1A_AE[k]+r2A_AE[k])

    logit(p15B_AE[k]) <- logit(p15A_AE[k]) + log(OR15[k])
    logit(p35B_AE[k]) <- min( logit(p35A_AE[k]) + log(OR35[k]), logit(p15B_AE[k]) )
  }

  for (i in 1:n_study){
    u_study1[i] ~ dnorm(mu_study1, pow(tau_study1,-2))
    u_study2[i] ~ dnorm(mu_study2, pow(tau_study2,-2))
  }

  for (j in 1:(J1_A+J2_A+J3_A+J4_A)){  
    p_A.adj[j,1] <- max(1e-299, min(1-p15A[j], 1-(1e-299))) 
    p_A.adj[j,2] <- max(p_A.adj[j,1]+1e-300, min(1-p35A[j], 1-(1e-300))) - p_A.adj[j,1]
    p_A.adj[j,3] <- 1 - max(p_A.adj[j,1]+1e-300, min(1-p35A[j], 1-(1e-300)))

    r1A[j] <- exp( u_study1[study_A[j]] + u_INT1[study_A[j], AE_A[j]] )
    r2A[j] <- exp( u_study2[study_A[j]] + u_INT2[study_A[j], AE_A[j]] )
    p35A[j] <- r2A[j]/(1+r1A[j]+r2A[j])
    p15A[j] <- (r1A[j]+r2A[j])/(1+r1A[j]+r2A[j])
  }
  for (j in 1:(J1_B+J2_B+J3_B+J4_B)){
    p_B.adj[j,1] <- max(1e-299, min(1-p15B[j], 1-(1e-299))) 
    p_B.adj[j,2] <- max(p_B.adj[j,1]+1e-300, min(1-p35B[j], 1-(1e-300))) - p_B.adj[j,1]
    p_B.adj[j,3] <- 1 - max(p_B.adj[j,1]+1e-300, min(1-p35B[j], 1-(1e-300)))

    r1A_mean[j] <- exp( u_study1[study_B[j]] + u_INT1[study_B[j], AE_B[j]] )
    r2A_mean[j] <- exp( u_study2[study_B[j]] + u_INT2[study_B[j], AE_B[j]] )
    logit(p15B[j]) <- logit((r1A_mean[j]+r2A_mean[j])/(1+r1A_mean[j]+r2A_mean[j])) + log(OR15[AE_B[j]])
    logit(p35B[j]) <- min( logit(r2A_mean[j]/(1+r1A_mean[j]+r2A_mean[j])) + log(OR35[AE_B[j]]), logit(p15B[j]) )
  }  
 
  # Case 1: both severity outcomes are fully observed
  for (j in 1:J1_A){  
    Y_A[j,1:3] ~ dmulti(p_A.adj[j,1:3], N_A[j])
  }
  for (j in 1:J1_B){
    Y_B[j,1:3] ~ dmulti(p_B.adj[j,1:3], N_B[j])
  }

  # Case 2: grade 1-5 observed; grade 3-5 left-censored
  for (j in 1:J2_A){  
    zeros_10_A[j] ~ dpois( -log(max(cumu2A_p35[j], 1e-300)) )  
    cumu2A_p35[j] <- pbin(cut35_A[J1_A+j], p_A.adj[J1_A+j,3]/(p_A.adj[J1_A+j,2]+p_A.adj[J1_A+j,3]), N_A[J1_A+j]-Y_A[J1_A+j, 1])
    Y_A[J1_A+j, 1] ~ dbin(p_A.adj[J1_A+j,1], N_A[J1_A+j])
  }
  for (j in 1:J2_B){  
    zeros_10_B[j] ~ dpois( -log(max(cumu2B_p35[j], 1e-300)) )  
    cumu2B_p35[j] <- pbin(cut35_B[J1_B+j], p_B.adj[J1_B+j,3]/(p_B.adj[J1_B+j,2]+p_B.adj[J1_B+j,3]), N_B[J1_B+j]-Y_B[J1_B+j, 1])
    Y_B[J1_B+j, 1] ~ dbin(p_B.adj[J1_B+j,1], N_B[J1_B+j])
  }

  # Case 3: grade 1-5 left-censored (G0 right-censored); grade 3-5 observed
  for (j in 1:J3_A){  
    zeros_01_A[j] ~ dpois( -log(max(1-cumu3A_p0[j], 1e-300)) )  
    cumu3A_p0[j] <- pbin(N_A[J1_A+J2_A+j]-cut15_A[J1_A+J2_A+j]-1, p_A.adj[J1_A+J2_A+j,1]/(p_A.adj[J1_A+J2_A+j,1]+p_A.adj[J1_A+J2_A+j,2]), N_A[J1_A+J2_A+j]-Y_A[J1_A+J2_A+j, 3])
    Y_A[J1_A+J2_A+j, 3] ~ dbin(p_A.adj[J1_A+J2_A+j,3], N_A[J1_A+J2_A+j])
  }
  for (j in 1:J3_B){  
    zeros_01_B[j] ~ dpois( -log(max(1-cumu3B_p0[j], 1e-300)) ) 
    cumu3B_p0[j] <- pbin(N_B[J1_B+J2_B+j]-cut15_B[J1_B+J2_B+j]-1, p_B.adj[J1_B+J2_B+j,1]/(p_B.adj[J1_B+J2_B+j,1]+p_B.adj[J1_B+J2_B+j,2]), N_B[J1_B+J2_B+j]-Y_B[J1_B+J2_B+j, 3])
    Y_B[J1_B+J2_B+j, 3] ~ dbin(p_B.adj[J1_B+J2_B+j,3], N_B[J1_B+J2_B+j])
  }

  # Case 4: both severity outcomes are left-censored
  for (j in 1:J4_A){  
    zeros_00_A[j] ~ dpois( -loglik_00_A[j] )
    for (k in 0:cut35_A[J1_A+J2_A+J3_A+j]){ ## go over y35_cens
      logcomp1_A[j, k+1] <- logfact(N_A[J1_A+J2_A+J3_A+j])-logfact(k)-logfact(N_A[J1_A+J2_A+J3_A+j]-k)+k*log(p_A.adj[J1_A+J2_A+J3_A+j,3])+(N_A[J1_A+J2_A+J3_A+j]-k)*log(1-p_A.adj[J1_A+J2_A+J3_A+j,3]) 
      logcomp2_A[j, k+1] <- log( max(1-pbin(N_A[J1_A+J2_A+J3_A+j]-cut15_A[J1_A+J2_A+J3_A+j]-1, p_A.adj[J1_A+J2_A+J3_A+j,1]/(p_A.adj[J1_A+J2_A+J3_A+j,1]+p_A.adj[J1_A+J2_A+J3_A+j,2]), N_A[J1_A+J2_A+J3_A+j]-k), 1e-300) )
      lik_00_A_mat[j, k+1] <- exp( logcomp1_A[j, k+1] + logcomp2_A[j, k+1] ) 
    }
    loglik_00_A[j] <- log( max(sum(lik_00_A_mat[j, 1:(cut35_A[J1_A+J2_A+J3_A+j]+1)]), 1e-300) )
  }
  for (j in 1:J4_B){  
    zeros_00_B[j] ~ dpois( -loglik_00_B[j] )
    for (k in 0:cut35_B[J1_B+J2_B+J3_B+j]){ ## go over y35_cens
      logcomp1_B[j, k+1] <- logfact(N_B[J1_B+J2_B+J3_B+j])-logfact(k)-logfact(N_B[J1_B+J2_B+J3_B+j]-k)+k*log(p_B.adj[J1_B+J2_B+J3_B+j,3])+(N_B[J1_B+J2_B+J3_B+j]-k)*log(1-p_B.adj[J1_B+J2_B+J3_B+j,3]) 
      logcomp2_B[j, k+1] <- log( max(1-pbin(N_B[J1_B+J2_B+J3_B+j]-cut15_B[J1_B+J2_B+J3_B+j]-1, p_B.adj[J1_B+J2_B+J3_B+j,1]/(p_B.adj[J1_B+J2_B+J3_B+j,1]+p_B.adj[J1_B+J2_B+J3_B+j,2]), N_B[J1_B+J2_B+J3_B+j]-k), 1e-300) )
      lik_00_B_mat[j, k+1] <- exp( logcomp1_B[j, k+1] + logcomp2_B[j, k+1] ) 
    }
    loglik_00_B[j] <- log( max(sum(lik_00_B_mat[j, 1:(cut35_B[J1_B+J2_B+J3_B+j]+1)]), 1e-300) )
  }

}
